b13_stale
CREATE OR REPLACE FUNCTION public.b13_stale(g approval_grants)
 RETURNS boolean
 LANGUAGE plpgsql
 SET search_path TO 'pg_catalog', 'public', 'pg_temp'
AS $function$
#variable_conflict use_column
DECLARE r approval_requests; d approval_decisions;
BEGIN
 SELECT * INTO d FROM approval_decisions WHERE id=g.approval_decision_id;
 SELECT * INTO r FROM approval_requests WHERE id=d.approval_request_id;
 IF NOT pr216_approver_active(d.decided_by,g.project_id) OR EXISTS(
 SELECT 1 FROM approval_requests q WHERE q.project_id=r.project_id AND q.approval_purpose=r.approval_purpose
 AND coalesce(q.scientific_conclusion_version_id,q.manuscript_version_id)<>coalesce(r.scientific_conclusion_version_id,r.manuscript_version_id)
 AND q.requested_at>r.requested_at) THEN RETURN true; END IF;
 IF r.manuscript_version_id IS NOT NULL AND r.approval_purpose NOT IN ('READY_FOR_SUBMISSION','PREPARE_SUBMISSION') THEN
 RAISE EXCEPTION 'B14 consent/effect dependency unavailable' USING ERRCODE='P16G1'; END IF;
 BEGIN
  PERFORM b13_release_gate(r,d.decided_by);
 EXCEPTION WHEN SQLSTATE 'P16T1' OR SQLSTATE 'P16R1' OR SQLSTATE 'P16G1' OR SQLSTATE 'P16C1' OR SQLSTATE 'P16A1' OR SQLSTATE 'P13S1' OR SQLSTATE 'P14B1' THEN RETURN true;
 END;
 RETURN false;
END $function$

b20_child_guard
CREATE OR REPLACE FUNCTION public.b20_child_guard()
 RETURNS trigger
 LANGUAGE plpgsql
 SET search_path TO 'pg_catalog', 'public', 'pg_temp'
AS $function$
    DECLARE parent novelty_assessments; owner_xmin xid;
    BEGIN
      IF NOT pr218_capture_prelocked() THEN RAISE EXCEPTION 'capture prelocks required' USING ERRCODE='P17D3'; END IF;
      SELECT * INTO parent FROM novelty_assessments WHERE id=NEW.novelty_assessment_id;
      SELECT xmin INTO owner_xmin FROM novelty_assessments WHERE id=NEW.novelty_assessment_id;
      IF parent.id IS NULL THEN RETURN NEW; END IF;
      IF owner_xmin IS DISTINCT FROM pg_current_xact_id()::xid THEN RAISE EXCEPTION 'sealed screening set' USING ERRCODE='P17D1'; END IF;
      IF NOT parent.screening_captured OR NEW.created_by IS DISTINCT FROM parent.created_by
        OR NEW.created_at IS DISTINCT FROM parent.created_at THEN
        RAISE EXCEPTION 'screening attribution mismatch' USING ERRCODE='P17D3'; END IF;
      RETURN NEW;
    END $function$
